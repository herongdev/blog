**三、**`Vue 3`**中的最佳实践**
`### 1.` **与**`Composition API`**结合使用**
javascript
import { ref, onMounted, onUnmounted } from 'vue'
export function useAnimation() {
  const frame = ref(0)
  let animationId = null
  function animate() {
    frame.value++
    animationId = requestAnimationFrame(animate)
  }
  onMounted(() => {
    animationId = requestAnimationFrame(animate)
  })
  onUnmounted(() => {
    cancelAnimationFrame(animationId)
  })
  return { frame }
}
