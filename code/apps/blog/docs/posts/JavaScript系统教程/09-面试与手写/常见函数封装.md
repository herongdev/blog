检测变量类型

```
function typeOf(arg) {
return typeof arg !== 'object' ? typeof arg : Object.prototype.toString.call(arg);
}
console.log(typeOf([1,2]));
```

```
function typeOf(arg) {
if (typeof arg !== 'object') {
return typeof arg;
}
return Object.prototype.toString.call(arg);
}
console.log(typeOf([1,2]));
```
