# 计算一个组值在列表数组中的索引

> 本节目标：理解“计算一个组值在列表数组中的索引”的核心思路，并能把它用于实际开发或面试表达。
const columnIndexComputed = computed(\{
get: () =\> \{
let selectedValues = rangeComputed.value.map(dateStr =\> dateStr.split(/[- :]/).map(Number));
return pickerViewColumns.value.map((column, index) =\> \{
const columnData = pickerData.value[column];
const selectedValue = selectedValues[state.currentInputIndex][index];
return columnData.indexOf(selectedValue) ?? 0;
\});
\},
set: (newIndices) =\> \{
const newValues = newIndices.map((index, columnIndex) =\> \{
const column = pickerViewColumns.value[columnIndex];
return pickerData.value[column][index];
\});
// 此处需要将新的选中值转换为日期字符串，并更新modelValue
const newStartDate = newValues.slice(0, 3).join('-');
const newEndDate = newValues.slice(3, 6).join('-');
emit('update:modelValue', [newStartDate, newEndDate]);
\}
\});
