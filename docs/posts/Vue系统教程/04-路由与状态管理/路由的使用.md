//另外，Link必须被包含在Router内，所以最终App组件是在Router组件里的
import { HashRouter as Router, Route, Switch, Redirect } from 'reat-router-dom';
ReactDOM.render(
<Router>
<App>
<Switch>
<Route path='/' exact={true} component={Home} />
<Route path='/profile' component={Profile} />
<Route path='/user' component={User} />
<Redirect to='/' />
</Switch>
</App>
</Router>,

)
