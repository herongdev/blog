# ==或对象是进行字符串或数字判断

> 本节目标：理解“==或对象是进行字符串或数字判断”的核心思路，并能把它用于实际开发或面试表达。
if (this.isValidAxiosResponse(axios)) \{
const dataAxios = axios.data;
const \{ code, result, data, msg \} = dataAxios;
const statusMapping = \{
200: \{ status: 1, data: result || data \},
0: \{ status: 1, data \},
401: \{ status: 2, msg: '身份验证失败' \},
\};
const defaultStatus = \{ status: 3, msg: msg || '服务出错', data: null \};
this.result = statusMapping[code] || defaultStatus;
return this.result;
\}
