# 循环和异步

> 本节目标：理解“循环和异步”的核心思路，并能把它用于实际开发或面试表达。
```
for (var i = 0; i < 5; i++) {
(function () {
setTimeout(function () {
console.log(i);
}, i * 1000);
})(i);
}
for (var i = 0; i < 5; i++) {
    (function (i) {
        setTimeout(function () {
            console.log(i);
        }, i * 1000);
    })(i);
}
```

```
var arr = [];
for (var i = 0; i < 5; i++) {
(function (i) {
arr[i] = function() {
console.log(i);
}
})(i);
}
arr.forEach( item => item());
```

```
var val1 = 0, val2 = 0, val3 = 0;
for (var i = 1; i <= 3; i++) {
var i2 = i;
(function () {
var i3 = i;
setTimeout(function () {
val1 += i;
val2 += i2;
val3 += i3;
}, 0);
})();
}
setTimeout(function () {
console.log(val1, val2, val3);
}, 100);
```
