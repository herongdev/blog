# 循环，异步，闭包

> 本节目标：理解“循环，异步，闭包”的核心思路，并能把它用于实际开发或面试表达。
```
var val1 = 0, val2 = 0, val3 = 0;
for (var i = 1; i <= 3; i++) {
var i2 = i;
console.log('i2='+ i2);
(function () {
var i3 = i;
console.log('i3='+ i3);
setTimeout(function () {
val1 += i;
val2 += i2;
val3 += i3;
console.log('val1='+val1);
console.log('val2='+val2);
console.log('val3='+val3);
}, 0);
})();
}
setTimeout(function () {
console.log(val1, val2, val3);
}, 100);
```

```
var name = '222';
var a = {
name: '111',
say: function() {
console.log(this.name);
}
}
var fun = a.say;
fun();//
a.say();//
var b = {
name: '333',
say: function(fun) {
fun();
}
}
b.say(a.say);//
b.say = a.say;
b.say();//
```
